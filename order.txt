// Function for order page
function placeOrder() {
    let items = document.querySelectorAll("input[type=checkbox]:checked");

    let orderList = [];
    let total = 0;

    items.forEach(item => {
        let name = item.value;
        let price = parseInt(item.getAttribute("data-price"));
        orderList.push(name);
        total += price;
    });

    if (orderList.length === 0) {
        alert("Please select at least one item");
        return false;
    }
    // Store data
    localStorage.setItem("items", orderList.join(", "));
    localStorage.setItem("total", total);

    // Redirect to bill page
    window.location.href = "bill.html";
    return false;
}
// Display bill on bill.html
window.onload = function () {
    if (document.getElementById("items")) {
        document.getElementById("items").innerHTML =
            "Items: " + localStorage.getItem("items");
        document.getElementById("total").innerHTML =
            "Total Amount: ₹" + localStorage.getItem("total");
    }
};
// Feedback
function submitFeedback() {
    let fb = document.getElementById("feedback").value;
    if (fb === "") {
        alert("Please enter feedback");
    } else {
        alert("Thank you for your feedback!");
    }
}
